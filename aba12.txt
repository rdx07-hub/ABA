import numpy as np
import matplotlib.pyplot as plt
def dice(n):
    rolls=[]
    for i in range(n):
        two_dice=(np.random.randint(1,7)+np.random.randint(1,7))
        rolls.append(two_dice)
    return rolls

data=dice(99000)

Practical 6B   Apply the concept of expected value to evaluate different decision alternatives and select the optimal one.
def cal_expect_val(prb , payoffs):
    exp_val = 0
    for prob , payoff in zip (prb, payoffs):
        exp_val += prob * payoff
    return exp_val

def select_opt_decision_alter(dcs , exp_vals):
    opt_dc =dcs[0]
    max_exp_val = exp_vals[0]
    for dc , ev in zip(dcs, exp_vals):
        if ev > max_exp_val:
            opt_dc = dc
            max_exp_val = ev
    return opt_dc

prb=[0.3 , 0.5 , 0.2]
payoffs = [[1000, 2000 , 3000] ,
                 [ 1500, 1800, 2500],
                [1200, 2200 , 2800]]
dcs = ['A' , 'B' , 'C' ]

exp_vals = []
for payoff in payoffs:
    exp_val = cal_expect_val(prb ,payoff)
    exp_vals.append(exp_val)
opt_dc = select_opt_decision_alter(dcs , exp_vals)
print("Expected Values: " , exp_vals)
print("Optimal Decision : " , opt_dc)

Practical 10B - Apply Time Series Analysis Techniques (e.g., ARIMA, exponential smoothing) to forecast future demand for a product or services, and evaluate the accuracy.Time Series Forecasting using ARIMA and Exponential Smoothing.
   
import pandas as pd
from sklearn.metrics import mean_absolute_error, mean_squared_error
from statsmodels.tsa.arima.model import ARIMA
from statsmodels.tsa.holtwinters import ExponentialSmoothing
import numpy as np
import warnings
warnings.filterwarnings("ignore")
   
data = {
       'Month': ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'],
       'Demand': [120, 135, 150, 165, 180, 190, 210, 225, 240, 255, 270, 290]
   }
   
df = pd.DataFrame(data)
   
print("Dataset")
print(df)
   
train = df['Demand'][:10]
test = df['Demand'][10:]
   
arima_model = ARIMA(train, order=(1, 1, 1))
arima_fit = arima_model.fit()
arima_forecast = arima_fit.forecast(steps=len(test))
   
print("\nARIMA Forecast")
print(arima_forecast)
   
exp_model = ExponentialSmoothing(train, trend='add')
exp_fit = exp_model.fit()
exp_forecast = exp_fit.forecast(len(test))
   
print("\nExponential Smoothing Forecast")
print(exp_forecast)
   
print("\nActual Demand")
print(test.values)
   
mae_arima = mean_absolute_error(test, arima_forecast)
rmse_arima = np.sqrt(mean_squared_error(test, arima_forecast))
   
mae_exp = mean_absolute_error(test, exp_forecast)
rmse_exp = np.sqrt(mean_squared_error(test, exp_forecast))
   
print("\nARIMA Accuracy")
print("MAE :", round(mae_arima, 2))
print("RMSE :", round(rmse_arima, 2))
   
print("\nExponential Smoothing Accuracy")
print("MAE :", round(mae_exp, 2))
print("RMSE :", round(rmse_exp, 2))
   
if rmse_arima < rmse_exp:
   print("\nARIMA gives better forecasting accuracy.")
else:
   print("\nExponential Smoothing gives better forecasting accuracy.")

Practical 12: Analysis of Variance and Experimental Design:
A)Design and conduct an experiment to study the effects of different factors on a specific response variable, analyze the results using analysis of variance (ANOVA), and draw conclusions.

import pandas as pd
from scipy.stats import f_oneway

data = {
        "Fertilizer": ["A","A","A","A","A",
                       "B","B","B","B","B",
                       "C","C","C","C","C"],
        "Height": [20,22,21,23,24,
                   25,26,27,25,28,
                   30,31,29,32,30]
        }

df = pd.DataFrame(data)
print("Dataset")
print(df)

A= df[df["Fertilizer"]=="A"]["Height"]
B= df[df["Fertilizer"]=="B"]["Height"]
C= df[df["Fertilizer"]=="C"]["Height"]

F_statistic, p_value = f_oneway(A,B,C)

print("\n ANOVA Results")
print("F-Staistic =", round(F_statistic,2))
print("p_value =", round(p_value,6))

if p_value < 0.05:
    print("\n Conclusion")
    print("Reject the Null Hypothesis")
    print("Different fertilizers significantly affect plant growth")
else:
    print("\n Conclusion")
    print("Fail to Reject the Null Hypothesis.")
    print("No significant difference among fertilizers.")
